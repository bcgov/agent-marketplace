<div class="extension-prototype-detail">
  <a class="back-link" href="catalog.html?variant=direct">Back to Extensions</a>
  <div id="extension-detail" role="status" aria-live="polite">Loading Extension details…</div>
</div>
<script>
(() => {
  const host = document.getElementById('extension-detail');
  const id = new URLSearchParams(location.search).get('id');
  const escape = value => String(value ?? '').replace(/[&<>"']/g, char => (
    { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[char]
  ));
  if (!id) {
    host.textContent = 'No Extension selected. Return to the catalogue and choose one.';
    return;
  }
  const load = url => fetch(url).then(response => {
    if (!response.ok) throw new Error(`${url} could not be loaded`);
    return response.json();
  });
  const inline = text => escape(text).replace(/\[([^\]]+)\]\([^)]+\)/g, '$1')
    .replace(/`([^`]+)`/g, '<code>$1</code>')
    .replace(/\*\*([^*]+)\*\*/g, '<strong>$1</strong>');
  const list = (items, tag) => items.length
    ? `<${tag}>${items.map(item => `<li>${inline(item)}</li>`).join('')}</${tag}>`
    : '';
  Promise.all([load('assets/catalog.json'), load('assets/skill-previews-prototype.json')])
    .then(([data, previews]) => {
      const record = data.extensions.find(item => item.id === id);
      if (!record) {
        host.textContent = 'Extension not found in the published catalogue. Return to the catalogue to choose another.';
        return;
      }
      const preview = previews[id];
      if (!preview || preview.digest !== record.files?.find(item => item.path === 'SKILL.md')?.sha256) {
        throw new Error(`Checked SKILL.md preview is missing or outdated for ${id}`);
      }
      const source = record.source || {};
      const review = record.review || {};
      const policy = record.policy || {};
      const revision = source.revision || review.revision;
      const repository = source.repository;
      const path = source.path;
      const sourceLink = repository && revision && path ? `${repository}/blob/${revision}/${path}/SKILL.md` : '';
      const sourceTree = repository && revision && path ? `${repository}/tree/${revision}/${path}` : '';
      const historyLink = repository && path ? `${repository}/commits/main/${path}/SKILL.md` : '';
      const status = review.status === 'domain-reviewed'
        ? 'Domain reviewed at this revision'
        : review.status === 'marketplace-reviewed'
          ? 'Reviewed at this revision'
          : review.status === 'auto-scanned'
            ? 'Automated only; no human review published'
            : 'No review evidence published';
      const findings = policy.findings || [];
      const capabilities = record.capabilities || {};
      const declared = [
        ['Commands', capabilities.commands],
        ['Network', capabilities.network],
        ['Writes', capabilities.filesystem?.write],
        ['Reads', capabilities.filesystem?.read],
        ['Data classes', capabilities['data-classes']],
        ['Secrets', capabilities['secret-names']],
        ['Environment', capabilities['environment-variables']]
      ].filter(([, values]) => values?.length);
      const installable = record.type === 'skill' && record.lifecycle === 'active'
        && /^https:\/\/github\.com\/[^/]+\/[^/]+$/.test(repository || '') && path && revision;
      const repoPath = repository?.replace(/^https:\/\/github\.com\//, '');
      const commands = installable ? {
        apm: `apm install ${repoPath}/${path}#${revision} --target copilot`,
        npx: `npx skills@${data['installer-version']} add \\\n  ${sourceTree} \\\n  --agent github-copilot -y`
      } : {};
      const related = data.extensions
        .filter(item => item.id !== id && item.specialty === record.specialty)
        .slice(0, 3);
      const advisoryCount = findings.filter(finding => finding.severity === 'advisory').length;
      const result = policy.outcome === 'passed'
        ? `Marketplace policy passed${advisoryCount ? ` · ${advisoryCount} ${advisoryCount === 1 ? 'advisory' : 'advisories'}` : ''}`
        : 'No passing automated check published';
      host.removeAttribute('role');
      host.removeAttribute('aria-live');
      host.innerHTML = `
        <h1>${escape(record['display-name'])}</h1>
        <p class="detail-intro">${escape(record.summary || '')}</p>
        <div class="detail-grid">
          <div class="detail-main">
            <section aria-labelledby="preview-title">
              <div class="preview-heading">
                <h2 id="preview-title">Inside the skill</h2>
                ${sourceLink ? `<a href="${escape(sourceLink)}">Read the full SKILL.md</a>` : ''}
              </div>
              <div class="skill-preview">
                <h3>${escape(preview.title)}</h3>
                ${preview.intro ? `<p>${inline(preview.intro)}</p>` : ''}
                ${preview.useWhen.length ? `<h4>Use when</h4>${list(preview.useWhen.slice(0, 2), 'ul')}` : ''}
                <details><summary>View more of this skill</summary>
                  ${preview.useWhen.length > 2 ? list(preview.useWhen.slice(2), 'ul') : ''}
                  ${preview.dontUseWhen.length ? `<h4>Do not use when</h4>${list(preview.dontUseWhen, 'ul')}` : ''}
                  ${preview.workflow.length ? `<h4>First steps</h4>${list(preview.workflow, 'ol')}` : ''}
                  <p class="rail-note">Excerpt from the checked SKILL.md. Read the full file for all steps and references.</p>
                </details>
              </div>
            </section>
            <section aria-labelledby="prerequisites-title">
              <h2 id="prerequisites-title">Before you use it</h2>
              ${list(record.prerequisites || [], 'ul') || '<p>No prerequisites declared.</p>'}
            </section>
          </div>
          <aside class="detail-rail" aria-label="Install and review">
            <section aria-labelledby="review-title">
              <h2 id="review-title">Review at a glance</h2>
              <p class="review-result">${escape(result)}</p>
              <div class="scan-badges" aria-label="Scanner status">
                <span>${policy.outcome === 'passed' ? 'Policy: passed' : 'Policy: no pass'}</span>
                <span class="planned">Cisco skill-scanner: planned</span>
                <span class="planned">NVIDIA SkillSpector: planned</span>
              </div>
              <p class="rail-note">Planned scanners have no results yet.</p>
              <p>${escape(status)}</p>
              <p>Access level: ${escape(review.risk || 'Not available')}</p>
              ${advisoryCount ? `<details><summary>Read ${advisoryCount} ${advisoryCount === 1 ? 'advisory' : 'advisories'}</summary><ul>${findings.filter(finding => finding.severity === 'advisory').map(finding => `<li>${escape(finding.message)}</li>`).join('')}</ul></details>` : ''}
              ${declared.length ? `<details><summary>Declared access</summary><ul class="access-summary">${declared.map(([name, values]) => `<li><strong>${name}:</strong> ${escape(values.join(', '))}</li>`).join('')}</ul></details>` : ''}
              <details><summary>Publication details</summary>
                <dl>
                  <dt>Publisher</dt><dd>${escape(record.publisher || 'Not published')}</dd>
                  <dt>Maintainers</dt><dd>${escape((record.maintainers || []).join(', ') || 'Not published')}</dd>
                  ${review.reviewers?.length ? `<dt>Reviewers</dt><dd>${escape(review.reviewers.join(', '))}</dd>` : ''}
                  <dt>License</dt><dd>${escape(record.license || 'Not published')}</dd>
                  <dt>Revision</dt><dd>${escape(revision || 'Not available')}</dd>
                </dl>
                ${historyLink ? `<p><a href="${escape(historyLink)}">Source history</a></p>` : ''}
                <p class="rail-note">History is not a scan archive. Maintainers are not verified authors.</p>
              </details>
            </section>
            <section class="install-card" aria-labelledby="install-title">
              <h2 id="install-title">Install</h2>
              ${installable
                ? `<div class="install-switch" role="group" aria-label="Choose installation method">
                    <button type="button" class="shell-option active" aria-pressed="true" data-installer="apm">APM</button>
                    <button type="button" class="shell-option" aria-pressed="false" data-installer="npx">npx</button>
                    <button type="button" class="shell-option" aria-pressed="false" data-installer="manual">Manual</button>
                  </div>
                  <div id="installer-command"><pre><code id="extension-install-command">${escape(commands.apm)}</code></pre><button type="button" id="extension-copy">Copy command</button><p id="extension-copy-status" role="status"></p></div>
                  <div id="installer-manual" hidden>
                    <p><a href="${escape(`${repository}/archive/${revision}.zip`)}">Download the checked revision</a>, then copy the entire <code>${escape(path)}</code> folder into your workspace's <code>.agents/skills/</code> folder.</p>
                    <p><a href="${escape(sourceTree)}">Inspect the source folder</a> before copying. Manual installation does not create an APM lockfile.</p>
                  </div>
                  <p class="rail-note install-note" id="install-rationale">APM records the resolved source in a manifest and lockfile.</p>`
                : '<p>Installation is not available for this Extension.</p>'}
              <p class="rail-note">Use the checked revision. Confirm before installing.</p>
            </section>
          </aside>
        </div>
        ${related.length ? `<section class="detail-main" aria-labelledby="related-title">
          <h2 id="related-title">Other ${escape(record.specialty)} skills</h2>
          <ul class="related-list">${related.map(item => `<li>
            <a href="catalog-detail-prototype.html?id=${encodeURIComponent(item.id)}">${escape(item['display-name'])}</a>
            <span>${escape(item.summary || '')}</span>
          </li>`).join('')}</ul>
        </section>` : ''}
        <section class="activity-demo" aria-labelledby="activity-title">
          <h2 id="activity-title">Popularity signals to compare</h2>
          <p>Your activity in this browser only. These are working interaction counters, not site-wide popularity.</p>
          <div class="activity-counts" aria-live="polite">
            <div><strong id="activity-views">—</strong>Detail views</div>
            <div><strong id="activity-copies">—</strong>Command copies</div>
          </div>
          <p class="rail-note" id="activity-note">Shared totals need approved analytics or a service. Neither signal proves an install.</p>
        </section>`;
      document.title = `${record['display-name']} | BC Gov Agent Marketplace`;
      const activityKey = kind => `catalogue-prototype:${id}:${kind}`;
      const updateActivity = kind => {
        try {
          const key = activityKey(kind);
          const previous = Number(localStorage.getItem(key) || '0');
          if (!Number.isSafeInteger(previous) || previous < 0) throw new Error('Activity count is invalid');
          const count = previous + 1;
          localStorage.setItem(key, String(count));
          host.querySelector(`#activity-${kind}`).textContent = String(count);
        } catch (error) {
          console.warn('Browser-only activity could not be recorded', error);
          host.querySelector('#activity-note').textContent = 'Browser-only activity is unavailable here. Shared totals still require approved analytics or a service.';
        }
      };
      for (const kind of ['views', 'copies']) {
        try {
          const value = Number(localStorage.getItem(activityKey(kind)) || '0');
          host.querySelector(`#activity-${kind}`).textContent = Number.isSafeInteger(value) && value >= 0 ? String(value) : '—';
        } catch (error) {
          console.warn('Browser-only activity could not be read', error);
          host.querySelector('#activity-note').textContent = 'Browser-only activity is unavailable here. Shared totals still require approved analytics or a service.';
        }
      }
      updateActivity('views');
      const installerNotes = {
        apm: 'APM records the resolved source in a manifest and lockfile.',
        npx: 'Skills CLI installs from the immutable tree. It does not create an APM lockfile.',
        manual: 'Copy every file from the pinned skill folder. Review the content first.'
      };
      for (const tab of host.querySelectorAll('[data-installer]')) {
        tab.addEventListener('click', () => {
          const method = tab.dataset.installer;
          host.querySelector('#installer-command').hidden = method === 'manual';
          host.querySelector('#installer-manual').hidden = method !== 'manual';
          if (method !== 'manual') host.querySelector('#extension-install-command').textContent = commands[method];
          host.querySelector('#install-rationale').textContent = installerNotes[method];
          host.querySelector('#extension-copy-status').textContent = '';
          for (const other of host.querySelectorAll('[data-installer]')) {
            const active = other === tab;
            other.classList.toggle('active', active);
            other.setAttribute('aria-pressed', String(active));
          }
        });
      }
      host.querySelector('#extension-copy')?.addEventListener('click', () => {
        const command = host.querySelector('#extension-install-command').textContent;
        navigator.clipboard.writeText(command)
          .then(() => {
            host.querySelector('#extension-copy-status').textContent = 'Install command copied.';
            updateActivity('copies');
          })
          .catch(() => { host.querySelector('#extension-copy-status').textContent = 'Could not copy the command. Select and copy the text above.'; });
      });
    })
    .catch(error => {
      console.error('Extension details could not be loaded', error);
      host.textContent = 'Catalogue or checked skill preview could not be loaded. Reload this page or return to the catalogue.';
    });
})();
</script>
