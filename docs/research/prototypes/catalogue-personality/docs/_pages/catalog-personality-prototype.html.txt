<script>
  (() => {
    const names = { A: 'A / Field guide', B: 'B / Release desk', C: 'C / Specialty workbench', D: 'D / Browse + evidence' };
    const keys = Object.keys(names);
    const label = document.getElementById('prototype-current');
    let active = keys.includes(new URLSearchParams(location.search).get('variant'))
      ? new URLSearchParams(location.search).get('variant') : 'A';

    function show(key) {
      active = key;
      document.querySelectorAll('.prototype-variant').forEach(section => {
        section.hidden = section.dataset.variant !== key;
      });
      label.textContent = names[key];
      const url = new URL(location.href);
      url.searchParams.set('variant', key);
      history.replaceState(null, '', url);
    }
    function cycle(direction) {
      show(keys[(keys.indexOf(active) + direction + keys.length) % keys.length]);
    }
    document.getElementById('prototype-prev').addEventListener('click', () => cycle(-1));
    document.getElementById('prototype-next').addEventListener('click', () => cycle(1));
    document.addEventListener('keydown', event => {
      if (!['ArrowLeft', 'ArrowRight'].includes(event.key)) return;
      const target = event.target;
      if (target.closest('input, textarea, select, [contenteditable="true"]')) return;
      cycle(event.key === 'ArrowRight' ? 1 : -1);
    });
    window.addEventListener('popstate', () => {
      const key = new URLSearchParams(location.search).get('variant');
      show(keys.includes(key) ? key : 'A');
    });
    const search = document.getElementById('browse-search');
    const results = [...document.querySelectorAll('.browse-results li')];
    function filter() {
      const specialty = document.querySelector('[name="browse-specialty"]:checked').value;
      const query = search.value.trim().toLowerCase();
      let shown = 0;
      results.forEach(row => {
        row.hidden = (specialty && row.dataset.specialty !== specialty) || !row.dataset.search.includes(query);
        if (!row.hidden) shown++;
      });
      document.getElementById('browse-count').textContent = `${shown} example Extensions shown`;
      document.getElementById('browse-empty').hidden = shown !== 0;
    }
    search.addEventListener('input', filter);
    document.querySelectorAll('[name="browse-specialty"]').forEach(radio => radio.addEventListener('change', filter));
    filter();
    const examples = {
      azure: {
        title: 'Azure Networking', summary: 'Design BC Gov Azure Landing Zone subnets, network security groups, service delegations, and private endpoints.',
        policy: 'Passed at current revision; no policy findings published.', review: 'Auto-checked; no human review claimed.',
        maintainers: '@mishraomp, @DerekRoberts', revision: '2046da7b0de0',
        path: 'skills/community/azure-networking'
      },
      actions: {
        title: 'GitHub Actions', summary: 'Repository automation guidance.',
        policy: 'Inspect the current catalogue record.', review: 'Inspect the current catalogue record.',
        maintainers: 'See current catalogue record', revision: 'See current catalogue record',
        path: 'skills/community/github-actions'
      },
      hardening: {
        title: 'Repo Hardening', summary: 'Apply a baseline repository hardening recipe for branch protection, secret handling, review gates, and deterministic validation.',
        policy: 'Passed at current revision; no policy findings published.', review: 'Domain reviewed by @bcgov/security at the published revision.',
        maintainers: '@mishraomp, @DerekRoberts', revision: '2046da7b0de0',
        path: 'skills/security/repo-hardening'
      }
    };
    let selected = null;
    function activity(id, field, increment) {
      try {
        const key = `prototype-catalogue-${id}-${field}`;
        const count = Number(localStorage.getItem(key) || '0') + increment;
        if (increment) localStorage.setItem(key, String(count));
        return count;
      } catch (error) {
        document.getElementById('detail-local-count').textContent = 'Browser storage unavailable; local activity cannot be saved.';
        return null;
      }
    }
    function showActivity() {
      const opens = activity(selected, 'opens', 0);
      const copies = activity(selected, 'copies', 0);
      if (opens !== null && copies !== null) {
        document.getElementById('detail-local-count').textContent =
          `Your activity in this browser only: ${opens} opens, ${copies} copies. Not global installs.`;
      }
    }
    document.querySelectorAll('[data-detail]').forEach(button => button.addEventListener('click', () => {
      selected = button.dataset.detail;
      activity(selected, 'opens', 1);
      const item = examples[button.dataset.detail];
      document.getElementById('detail-title').textContent = item.title;
      document.getElementById('detail-summary').textContent = item.summary;
      for (const field of ['policy', 'review', 'maintainers', 'revision']) {
        document.getElementById(`detail-${field}`).textContent = item[field];
      }
      document.getElementById('detail-fields').hidden = false;
      const historyLink = document.getElementById('detail-history');
      historyLink.href = `https://github.com/bcgov/agent-marketplace/commits/main/${item.path}`;
      historyLink.hidden = false;
      document.getElementById('detail-copy').hidden = false;
      showActivity();
    }));
    document.getElementById('detail-copy').addEventListener('click', async () => {
      if (!selected) return;
      const item = examples[selected];
      const command = `apm install bcgov/agent-marketplace/${item.path}#2046da7b0de0f673bf39208b35385c1516be4c75 --target copilot`;
      try {
        await navigator.clipboard.writeText(command);
        activity(selected, 'copies', 1);
        showActivity();
      } catch (error) {
        document.getElementById('detail-local-count').textContent = 'Clipboard unavailable; no copy recorded.';
      }
    });
    show(active);
  })();
</script>
